package color

import (
    "os"
    "syscall"
    "unsafe"
)

var (
    kernel32         = syscall.NewLazyDLL("kernel32.dll")
    procGetConsoleMode = kernel32.NewProc("GetConsoleMode")
)

// IsTerminal 检查 stdout 是否连接到终端
func IsTerminal() bool {
    return isTerminal(os.Stdout.Fd())
}

func isTerminal(fd uintptr) bool {
    var st uint32
    r, _, _ := procGetConsoleMode.Call(fd, uintptr(unsafe.Pointer(&st)))
    return r != 0
}

// SupportsColor 检查终端是否支持颜色
func SupportsColor() bool {
    // Windows Terminal 和 ConEmu 支持 ANSI
    // 检查环境变量
    if os.Getenv("WT_SESSION") != "" {
        return true // Windows Terminal
    }
    if os.Getenv("ConEmuANSI") == "ON" {
        return true // ConEmu
    }
    // 检查是否是终端（传统 cmd 需要 ENABLE_VIRTUAL_TERMINAL_PROCESSING）
    return IsTerminal()
}

// shared/argparse/parser.go
package argparse

import "strings"

// Linux 独有选项（触发 Linux 模式）
var linuxOptions = map[string]bool{
    "l": true, "a": true, "h": true, "t": true,
    "R": true, "r": true, "1": true, "d": true,
    "F": true, "S": true, "i": true,
    "all": true, "color": true,
}

// ParseArgs 解析命令行参数
// 返回: (选项map, 位置参数列表)
func ParseArgs(args []string) (map[string]string, []string) {
    opts := make(map[string]string)
    var positional []string

    for i := 0; i < len(args); i++ {
        arg := args[i]

        if strings.HasPrefix(arg, "--") {
            // 长选项: --option 或 --option=value
            key := arg[2:]
            if idx := strings.Index(key, "="); idx != -1 {
                opts[key[:idx]] = key[idx+1:]
            } else {
                opts[key] = ""
            }
        } else if strings.HasPrefix(arg, "-") && len(arg) > 1 {
            // 短选项: -l, -a, -lah (组合)
            flags := arg[1:]
            for _, ch := range flags {
                opts[string(ch)] = ""
            }
        } else if strings.HasPrefix(arg, "/") && len(arg) == 2 {
            // Windows 风格: /W, /B
            opts[arg[1:]] = ""
        } else {
            // 位置参数
            positional = append(positional, arg)
        }
    }

    return opts, positional
}

// HasLinuxOption 检查参数中是否包含 Linux 独有选项
func HasLinuxOption(args []string) bool {
    for _, arg := range args {
        if strings.HasPrefix(arg, "--") {
            key := arg[2:]
            if idx := strings.Index(key, "="); idx != -1 {
                key = key[:idx]
            }
            if linuxOptions[key] {
                return true
            }
        } else if strings.HasPrefix(arg, "-") && len(arg) > 1 {
            flags := arg[1:]
            for _, ch := range flags {
                if linuxOptions[string(ch)] {
                    return true
                }
            }
        }
    }
    return false
}
```

// shared/osutil/fileinfo.go
package osutil

import (
    "fmt"
    "os"
    "path/filepath"
    "strings"
    "syscall"
    "unsafe"
)

var (
    kernel32                = syscall.NewLazyDLL("kernel32.dll")
    procGetFileInformationByHandle = kernel32.NewProc("GetFileInformationByHandle")
)

// IsExecutable 检查文件是否是可执行文件
func IsExecutable(name string) bool {
    ext := strings.ToLower(filepath.Ext(name))
    switch ext {
    case ".exe", ".bat", ".cmd", ".ps1", ".com", ".msi":
        return true
    }
    return false
}

// FormatSize 格式化文件大小（人类可读）
func FormatSize(size int64) string {
    const (
        KB = 1024
        MB = KB * 1024
        GB = MB * 1024
    )

    switch {
    case size >= GB:
        return fmt.Sprintf("%.1fG", float64(size)/float64(GB))
    case size >= MB:
        return fmt.Sprintf("%.1fM", float64(size)/float64(MB))
    case size >= KB:
        return fmt.Sprintf("%.1fK", float64(size)/float64(KB))
    default:
        return fmt.Sprintf("%dB", size)
    }
}

// GetFileMode 获取类似 Linux 的文件权限字符串
func GetFileMode(name string) string {
    info, err := os.Stat(name)
    if err != nil {
        return "?---------"
    }

    var mode strings.Builder

    // 文件类型
    if info.IsDir() {
        mode.WriteRune('d')
    } else if info.Mode()&os.ModeSymlink != 0 {
        mode.WriteRune('l')
    } else {
        mode.WriteRune('-')
    }

    // Windows 权限映射
    // 检查文件是否可读（几乎所有文件都可读）
    mode.WriteString("r--")

    // 检查文件是否可写
    if info.Mode().Perm()&0200 != 0 {
        mode.WriteString("w-")
    } else {
        mode.WriteString("--")
    }

    // 检查文件是否可执行
    if IsExecutable(name) {
        mode.WriteString("x")
    } else {
        mode.WriteString("-")
    }

    // 重复三次（owner, group, other）
    mode.WriteString(mode.String()[1:])

    return mode.String()[:10]
}

// GetFileOwner 获取文件所有者（Windows）
func GetFileOwner(name string) string {
    // 简化实现：返回当前用户名
    return os.Getenv("USERNAME")
}

// GetFileGroup 获取文件组（Windows）
func GetFileGroup(name string) string {
    // Windows 没有组概念，返回空或用户组
    return os.Getenv("USERDOMAIN")
}

// IsHidden 检查文件是否隐藏
func IsHidden(name string) bool {
    // 检查文件名是否以 . 开头
    base := filepath.Base(name)
    if strings.HasPrefix(base, ".") {
        return true
    }

    // 检查 Windows 隐藏属性
    ptr, err := syscall.UTF16PtrFromString(name)
    if err != nil {
        return false
    }

    attrs, err := syscall.GetFileAttributes(ptr)
    if err != nil {
        return false
    }

    return attrs&syscall.FILE_ATTRIBUTE_HIDDEN != 0
}

// GetInode 获取文件索引号（Windows 用文件索引代替）
func GetInode(name string) uint64 {
    handle, err := syscall.CreateFile(
        syscall.StringToUTF16Ptr(name),
        0,
        syscall.FILE_SHARE_READ|syscall.FILE_SHARE_WRITE|syscall.FILE_SHARE_DELETE,
        nil,
        syscall.OPEN_EXISTING,
        syscall.FILE_FLAG_BACKUP_SEMANTICS,
        0,
    )
    if err != nil {
        return 0
    }
    defer syscall.CloseHandle(handle)

    var info struct {
        FileIndex      uint64
        FileIndexHigh  uint64
    }

    ret, _, _ := procGetFileInformationByHandle.Call(
        uintptr(handle),
        uintptr(unsafe.Pointer(&info)),
    )

    if ret == 0 {
        return 0
    }

    return info.FileIndex
}
```

package ls

import (
    "fmt"
    "os"
    "path/filepath"

    "github.com/user/win-tools/shared/argparse"
)

// Options ls 命令选项
type Options struct {
    // 通用选项
    Paths []string

    // Windows 模式选项
    WindowsMode bool
    Wide        bool // /W
    Bare        bool // /B
    Recursive   bool // /S
    SortBy      string // /O
    AttrFilter  string // /A

    // Linux 模式选项
    Long        bool // -l
    All         bool // -a, --all
    Human       bool // -h
    TimeSort    bool // -t
    Reverse     bool // -r
    Color       string // --color
    OnePerLine  bool // -1
    DirOnly     bool // -d
    FileType    bool // -F
    SizeSort    bool // -S
    Inode       bool // -i
}

// ParseOptions 从参数解析选项
func ParseOptions(args []string) Options {
    opts, positional := argparse.ParseArgs(args)

    options := Options{
        Paths: positional,
    }

    // 检查是否使用 Linux 模式
    if argparse.HasLinuxOption(args) {
        // Linux 模式
        options.Long = opts["l"] != "" || opts["long"] != ""
        options.All = opts["a"] != "" || opts["all"] != ""
        options.Human = opts["h"] != "" || opts["human-readable"] != ""
        options.TimeSort = opts["t"] != "" || opts["time"] != ""
        options.Reverse = opts["r"] != "" || opts["reverse"] != ""
        options.OnePerLine = opts["1"] != ""
        options.DirOnly = opts["d"] != ""
        options.FileType = opts["F"] != "" || opts["classify"] != ""
        options.SizeSort = opts["S"] != "" || opts["size"] != ""
        options.Inode = opts["i"] != "" || opts["inode"] != ""

        // Windows 模式
        options.WindowsMode = true
        options.Wide = opts["W"] != ""
        options.Bare = opts["B"] != ""
        options.Recursive = opts["S"] != ""
        options.SortBy = opts["O"]
        options.AttrFilter = opts["A"]
    }

    // 默认路径为当前目录
    if len(options.Paths) == 0 {
        options.Paths = []string{"."}
    }

            output := FormatWindows(path, options)
            fmt.Print(output)
        } else {
            output := FormatLinux(path, options)
            fmt.Print(output)
        }
    }

    return nil

// FormatWindows 生成 Windows dir 风格输出
func FormatWindows(dir string, opts Options) string {
    var sb strings.Builder

    absDir, err := filepath.Abs(dir)
    if err != nil {
        absDir = dir
    }

    // 获取驱动器信息
    drive := filepath.VolumeName(absDir)
    if drive == "" {
        drive = "C:"
    }

    sb.WriteString(fmt.Sprintf(" Volume in drive %s has no label\n", string(drive[0])))
    sb.WriteString(" Volume Serial Number is 0000-0000\n")
    sb.WriteString("\n")
    sb.WriteString(fmt.Sprintf(" Directory of %s\n\n", absDir))

    // 遍历目录
    entries, err := os.ReadDir(dir)
    if err != nil {
        sb.WriteString(fmt.Sprintf("Error reading directory: %v\n", err))
        return sb.String()
    }

    fileCount := 0
    dirCount := 0
    totalSize := int64(0)

    // 添加 . 和 ..
    sb.WriteString(formatWindowsEntry(".", true, 0, time.Time{}))
    sb.WriteString(formatWindowsEntry("..", true, 0, time.Time{}))
    dirCount += 2

    for _, entry := range entries {
        info, err := entry.Info()
        if err != nil {
            continue
        }

        isDir := entry.IsDir()
        if isDir {
            dirCount++
        } else {
            fileCount++
            totalSize += info.Size()
        }

        if opts.Bare {
            sb.WriteString(entry.Name() + "\n")
        } else if opts.Wide {
            if isDir {
                sb.WriteString(fmt.Sprintf("[%s]", entry.Name()))
            } else {
                sb.WriteString(entry.Name())
            }
            sb.WriteString("\t")
        } else {
            sb.WriteString(formatWindowsEntry(entry.Name(), isDir, info.Size(), info.ModTime()))
        }
    }

    if opts.Wide {
        sb.WriteString("\n")
    }

    sb.WriteString(fmt.Sprintf("%15d File(s) %15d bytes\n", fileCount, totalSize))
    sb.WriteString(fmt.Sprintf("%15d Dir(s)  %15d bytes free\n", dirCount, getFreeSpace(dir)))

    return sb.String()
}

func formatWindowsEntry(name string, isDir bool, size int64, modTime time.Time) string {
    if isDir {
        if modTime.IsZero() {
            return fmt.Sprintf("               <DIR>          %s\n", name)
        }
        return fmt.Sprintf("%s  %s    <DIR>          %s\n",
            modTime.Format("2006/01/02"), modTime.Format("15:04"), name)
    }
    return fmt.Sprintf("%s  %s %15d %s\n",
        modTime.Format("2006/01/02"), modTime.Format("15:04"), size, name)
}

func getFreeSpace(path string) int64 {
    // 获取磁盘可用空间
    var freeBytes int64
    kernel32 := syscall.NewLazyDLL("kernel32.dll")
    proc := kernel32.NewProc("GetDiskFreeSpaceExW")
    ret, _, _ := proc.Call(
        uintptr(unsafe.Pointer(syscall.StringToUTF16Ptr(path))),
        uintptr(unsafe.Pointer(&freeBytes)),
        0, 0,
    )
    if ret == 0 {
        return 0
    }
    return freeBytes
}

import (

// FormatLinux 生成 Linux ls 风格输出
func FormatLinux(dir string, opts Options) string {
    var sb strings.Builder

    entries, err := os.ReadDir(dir)
    if err != nil {
        return fmt.Sprintf("ls: cannot access '%s': %v\n", dir, err)
    }

    // 过滤隐藏文件
    var filtered []os.DirEntry
    for _, entry := range entries {
        if !opts.All && strings.HasPrefix(entry.Name(), ".") {
            continue
        }
        filtered = append(filtered, entry)
    }

    // 排序
    sortedEntries := sortEntries(filtered, dir, opts)

    // 输出
    for _, entry := range sortedEntries {
        fullPath := filepath.Join(dir, entry.Name())
        info, err := entry.Info()
        if err != nil {
            continue
        }

        if opts.Long {
            sb.WriteString(formatLinuxLong(entry.Name(), info, fullPath, opts))
        } else if opts.OnePerLine {
            sb.WriteString(colorizeName(entry.Name(), info, opts) + "\n")
        } else {
            sb.WriteString(colorizeName(entry.Name(), info, opts) + "  ")
        }
    }

    if !opts.Long && !opts.OnePerLine {
        sb.WriteString("\n")
    }

    return sb.String()
}

func sortEntries(entries []os.DirEntry, dir string, opts Options) []os.DirEntry {
    sorted := make([]os.DirEntry, len(entries))
    copy(sorted, entries)

    sort.Slice(sorted, func(i, j int) bool {
        if opts.TimeSort {
            infoI, _ := sorted[i].Info()
            infoJ, _ := sorted[j].Info()
            if opts.Reverse {
                return infoI.ModTime().Before(infoJ.ModTime())
            }
            return infoI.ModTime().After(infoJ.ModTime())
        }
        if opts.SizeSort {
            infoI, _ := sorted[i].Info()
            infoJ, _ := sorted[j].Info()
            if opts.Reverse {
                return infoI.Size() < infoJ.Size()
            }
            return infoI.Size() > infoJ.Size()
        }
        // 默认按名称排序
        if opts.Reverse {
            return sorted[i].Name() > sorted[j].Name()
        }
        return sorted[i].Name() < sorted[j].Name()
    })

    return sorted
}

func formatLinuxLong(name string, info os.FileInfo, fullPath string, opts Options) string {
    var sb strings.Builder

    // inode
    if opts.Inode {
        sb.WriteString(fmt.Sprintf("%d ", osutil.GetInode(fullPath)))
    }

    // 权限
    mode := osutil.GetFileMode(fullPath)
    sb.WriteString(mode + " ")

    // 链接数
    sb.WriteString("1 ")

    // 所有者
    sb.WriteString(osutil.GetFileOwner(fullPath) + " ")

    // 组
    sb.WriteString(osutil.GetFileGroup(fullPath) + " ")

    // 大小
    if opts.Human {
        sb.WriteString(fmt.Sprintf("%6s ", osutil.FormatSize(info.Size())))
    } else {
        sb.WriteString(fmt.Sprintf("%6d ", info.Size()))
    }

    // 修改时间
    sb.WriteString(info.ModTime().Format("Jan  2 15:04") + " ")

    // 文件名（带颜色）
    sb.WriteString(colorizeName(name, info, opts))

    // 文件类型标记
    if opts.FileType {
        if info.IsDir() {
            sb.WriteString("/")
        } else if osutil.IsExecutable(name) {
            sb.WriteString("*")
        } else if info.Mode()&os.ModeSymlink != 0 {
            sb.WriteString("@")
        }
    }

    sb.WriteString("\n")

    return sb.String()
}

func colorizeName(name string, info os.FileInfo, opts Options) string {
    if opts.Color == "never" || (opts.Color == "auto" && !color.SupportsColor()) {
        return name
    }

    if info.IsDir() {
        return color.Colorize(name, color.Blue)
    }
    if osutil.IsExecutable(name) {
        return color.Colorize(name, color.Green)
    }
    if info.Mode()&os.ModeSymlink != 0 {
        return color.Colorize(name, color.Cyan)
    }

    // 压缩文件
    ext := strings.ToLower(filepath.Ext(name))
    switch ext {
    case ".zip", ".gz", ".tar", ".rar", ".7z", ".bz2":
        return color.Colorize(name, color.Red)
    }

    // 图片文件
    switch ext {
    case ".jpg", ".jpeg", ".png", ".gif", ".bmp", ".svg":
        return color.Colorize(name, color.Magenta)
    }

    return name
}

// pkg/grep/grep.go
package grep

import (
    "bufio"
    "fmt"
    "io"
    "regexp"
    "strings"

    "github.com/user/win-tools/shared/color"
)

// Options grep 命令选项
type Options struct {
    Pattern     string
    IgnoreCase  bool
    LineNumber  bool
    Count       bool
    Invert      bool
    OnlyMatch   bool
    WordRegexp  bool
    ExtendedReg bool
    FixedString bool
    Color       string
}

// Grep 从 reader 读取并匹配模式
func Grep(reader io.Reader, pattern string, opts Options, writer io.Writer) error {
    // 编译正则表达式
    regexPattern := pattern
    if opts.FixedString {
        regexPattern = regexp.QuoteMeta(pattern)
    }
    if opts.WordRegexp {
        regexPattern = `\b` + regexPattern + `\b`
    }
    if opts.IgnoreCase {
        regexPattern = "(?i)" + regexPattern
    }

    re, err := regexp.Compile(regexPattern)
    if err != nil {
        return fmt.Errorf("invalid pattern: %v", err)
    }

    scanner := bufio.NewScanner(reader)
    lineNum := 0
    matchCount := 0

    for scanner.Scan() {
        lineNum++
        line := scanner.Text()
        matched := re.MatchString(line)

        if opts.Invert {
            matched = !matched
        }

        if matched {
            matchCount++

            if opts.Count {
                continue
            }

            var output string

            if opts.OnlyMatch {
                matches := re.FindAllString(line, -1)
                for _, m := range matches {
                    if opts.LineNumber {
                        fmt.Fprintf(writer, "%d:%s\n", lineNum, m)
                    } else {
                        fmt.Fprintf(writer, "%s\n", m)
                    }
                }
                continue
            }

            if opts.LineNumber {
                output = fmt.Sprintf("%d:", lineNum)
            }

            // 高亮匹配
            if opts.Color != "never" && opts.Color != "" {
                output += re.ReplaceAllString(line, color.Colorize("$0", color.Red))
            } else {
                output += line
            }

            fmt.Fprintln(writer, output)
        }
    }

    if opts.Count {
        fmt.Fprintf(writer, "%d\n", matchCount)
    }

    return scanner.Err()
}

package cat

import (
    "bufio"
    "fmt"
    "io"
)

// Options cat 命令选项
type Options struct {
    Number      bool // -n
    ShowEnds    bool // -E
    ShowTabs    bool // -T
    SqueezeBlank bool // -s
}

// Cat 从 reader 读取并输出到 writer
func Cat(reader io.Reader, opts Options, writer io.Writer) error {
    scanner := bufio.NewScanner(reader)
    lineNum := 0
    prevBlank := false

    for scanner.Scan() {
        lineNum++
        line := scanner.Text()

        // 压缩空行
        if opts.SqueezeBlank {
            isBlank := line == ""
            if isBlank && prevBlank {
                continue
            }
            prevBlank = isBlank
        }

        // 行号
        if opts.Number {
            fmt.Fprintf(writer, "%6d\t", lineNum)
        }

        // 显示制表符
        if opts.ShowTabs {
            line = strings.ReplaceAll(line, "\t", "^I")
        }

        fmt.Fprint(writer, line)

        // 显示行尾
        if opts.ShowEnds {
            fmt.Fprint(writer, "$")
        }

        fmt.Fprintln(writer)
    }

    return scanner.Err()
}

// pkg/which/which.go
package which

import (
    "fmt"
    "os"
    "path/filepath"
    "strings"
)

// Which 在 PATH 中查找可执行文件
func Which(name string) (string, error) {
    // 如果包含路径分隔符，直接检查
    if strings.ContainsAny(name, `/\`) {
        if _, err := os.Stat(name); err == nil {
            return name, nil
        }
        return "", fmt.Errorf("not found: %s", name)
    }

    // 获取 PATH 环境变量
    pathEnv := os.Getenv("PATH")
    paths := strings.Split(pathEnv, ";")

    // 可能的扩展名
    extensions := []string{"", ".exe", ".bat", ".cmd", ".ps1"}

    for _, dir := range paths {
        for _, ext := range extensions {
            fullPath := filepath.Join(dir, name+ext)
            if _, err := os.Stat(fullPath); err == nil {
                return fullPath, nil
            }
        }
    }

    return "", fmt.Errorf("not found in PATH: %s", name)
}

// WhichAll 查找所有匹配的可执行文件
func WhichAll(name string) ([]string, error) {
    var results []string

    if strings.ContainsAny(name, `/\`) {
        if _, err := os.Stat(name); err == nil {
            return []string{name}, nil
        }
        return nil, fmt.Errorf("not found: %s", name)
    }

    pathEnv := os.Getenv("PATH")
    paths := strings.Split(pathEnv, ";")
    extensions := []string{"", ".exe", ".bat", ".cmd", ".ps1"}

    for _, dir := range paths {
        for _, ext := range extensions {
            fullPath := filepath.Join(dir, name+ext)
            if _, err := os.Stat(fullPath); err == nil {
                results = append(results, fullPath)
            }
        }
    }

    if len(results) == 0 {
        return nil, fmt.Errorf("not found in PATH: %s", name)
    }

    return results, nil
}
```

package main

import (
    "fmt"
    "os"

    "github.com/user/win-tools/pkg/which"
)

var version = "dev"

func main() {
    if len(os.Args) > 1 && (os.Args[1] == "--version" || os.Args[1] == "-v") {
        println("win-tools which", version)
        os.Exit(0)
    }

    if len(os.Args) < 2 {
        fmt.Fprintf(os.Stderr, "Usage: which [-a] COMMAND...\n")
        os.Exit(2)
    }

    showAll := false
    var commands []string

    for _, arg := range os.Args[1:] {
        switch arg {
        case "-a", "--all":
            showAll = true
        default:
            commands = append(commands, arg)
        }
    }

    for _, cmd := range commands {
        if showAll {
            results, err := which.WhichAll(cmd)
            if err != nil {
                fmt.Fprintf(os.Stderr, "which: %s: %v\n", cmd, err)
                os.Exit(1)
            }
            for _, r := range results {
                fmt.Println(r)
            }
        } else {
            result, err := which.Which(cmd)
            if err != nil {
                fmt.Fprintf(os.Stderr, "which: %s: %v\n", cmd, err)
                os.Exit(1)
            }
            fmt.Println(result)
        }
    }
}

# install.ps1
param(
    [string]$InstallDir = "$env:LOCALAPPDATA\win-tools",
    [string]$BackupDir = "$env:LOCALAPPDATA\win-tools-backup",
    [switch]$AddToPath,
    [switch]$Force,
    [switch]$Restore
)

$ErrorActionPreference = "Stop"

# 恢复模式
if ($Restore) {
    if (-not (Test-Path $BackupDir)) {
        Write-Host "No backup directory found at: $BackupDir" -ForegroundColor Red
        exit 1
    }

    Write-Host "Restoring from backup..." -ForegroundColor Yellow

    Get-ChildItem $BackupDir -Directory | ForEach-Object {
        $cmdName = $_.Name
        $metadataPath = Join-Path $_.FullName "metadata.json"
        $backupFile = Get-ChildItem $_.FullName -Filter "*.original" | Select-Object -First 1

        if ($backupFile -and (Test-Path $metadataPath)) {
            $metadata = Get-Content $metadataPath | ConvertFrom-Json
            $originalPath = $metadata.OriginalPath

            Write-Host "Restoring $cmdName to: $originalPath" -ForegroundColor Cyan
            Copy-Item $backupFile.FullName $originalPath -Force
        }
    }

    # 删除安装目录
    Remove-Item -Recurse -Force $InstallDir -ErrorAction SilentlyContinue
    Write-Host "Restore complete!" -ForegroundColor Green
    exit 0
}

# 安装模式
if (-not (Test-Path ".\dist")) {
    Write-Host "dist directory not found. Run build.ps1 first." -ForegroundColor Red
    exit 1
}

Write-Host "Installing win-tools..." -ForegroundColor Cyan

# 创建目录
New-Item -ItemType Directory -Force $InstallDir | Out-Null
New-Item -ItemType Directory -Force $BackupDir | Out-Null

# 备份已有命令
$commands = Get-ChildItem ".\dist\*.exe" | ForEach-Object { $_.BaseName }

foreach ($cmd in $commands) {
    $existing = Get-Command $cmd -ErrorAction SilentlyContinue
    if ($existing) {
        $backupPath = Join-Path $BackupDir $cmd
        New-Item -ItemType Directory -Force $backupPath | Out-Null

        # 备份原始文件
        Copy-Item $existing.Source "$backupPath\$cmd.exe.original" -Force

        # 保存元信息
        @{
            OriginalPath = $existing.Source
            BackupDate = Get-Date -Format "yyyy-MM-dd HH:mm:ss"
            ToolVersion = (Get-Item $existing.Source).VersionInfo.FileVersion
        } | ConvertTo-Json | Set-Content "$backupPath\metadata.json"

        Write-Host "Backed up: $($existing.Source)" -ForegroundColor Yellow
    }
}

# 安装新命令
Copy-Item ".\dist\*.exe" $InstallDir -Force:$Force

# 添加到 PATH
if ($AddToPath) {
    $currentPath = [Environment]::GetEnvironmentVariable("Path", "User")
    if ($currentPath -notlike "*$InstallDir*") {
        [Environment]::SetEnvironmentVariable("Path", "$currentPath;$InstallDir", "User")
        Write-Host "Added to PATH: $InstallDir" -ForegroundColor Green
        Write-Host "Restart your terminal to apply changes." -ForegroundColor Yellow
    }
}

Write-Host "`nInstallation complete!" -ForegroundColor Green
Write-Host "Installed to: $InstallDir" -ForegroundColor Green
Write-Host "Backups at: $BackupDir" -ForegroundColor Green
Write-Host "`nTo restore original commands: .\install.ps1 -Restore" -ForegroundColor Gray
```
